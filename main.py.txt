from fastapi import FastAPI, HTTPException, Security, Depends
from fastapi.security.api_key import APIKeyHeader
from pydantic import BaseModel, HttpUrl
import os
from scraper import SimpleScraper
from security import verify_api_key
from ai_analyzer import analyzer

app = FastAPI(title="Cloud Web Scraper & AI Analyzer API")

class ScrapeRequest(BaseModel):
    url: HttpUrl

@app.get("/")
def read_root():
    return {
        "status": "online",
        "message": "Cloud Web Scraper & AI Analyzer API is running smoothly!"
    }

@app.post("/scrape")
def scrape_url(request: ScrapeRequest, api_key: str = Depends(verify_api_key)):
    try:
        scraper = SimpleScraper(str(request.url))
        scraped_data = scraper.scrape()
        
        # AI Insights generation
        ai_insights = analyzer.generate_marketing_insights(scraped_data)
        scraped_data["ai_marketing_insights"] = ai_insights
        
        return {
            "success": True,
            "data": scraped_data
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
